See it in action

Here's a Python example showing how easy it is to integrate Ranukita's API.

# Ranukita Scraping API - Quick Start Example
# Install: pip install requests

import requests
import json

# Your API key from ranuk.dev/dashboard
API_KEY = "rk_live_your_key_here"
BASE_URL = "https://api.ranuk.dev/v1"

# Scrape product listings from an e-commerce site
response = requests.post(
    f"{BASE_URL}/scrape",
    headers={"Authorization": f"Bearer {API_KEY}"},
    json={
        "target": "https://example-store.com/products",
        "format": "json",
        "fields": ["name", "price", "rating", "reviews", "image_url"],
        "pagination": True,
        "max_pages": 5
    }
)

data = response.json()

# Process structured results
for product in data["results"]:
    print(f"📦 {product['name']}")
    print(f"   💰 ${product['price']} | ⭐ {product['rating']}")
    print(f"   📝 {product['reviews']} reviews")
    print()

# Output:
# 📦 Wireless Noise-Canceling Headphones
#    💰 $79.99 | ⭐ 4.7
#    📝 2,341 reviews
#
# 📦 Smart Watch Pro 2026
#    💰 $199.99 | ⭐ 4.5
#    📝 1,892 reviews
